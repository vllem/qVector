cmake_minimum_required(VERSION 3.20)
project(qvector LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)

option(VC_DEMO "Include the fake homeserver so `qvector --demo` runs without an account (dev aid)" ON)

option(VC_CONSOLE "Build the engine with tokio-console support (dev aid: run `tokio-console` while the app runs)" OFF)
option(VC_PROFILE "Keep symbols and frame pointers so `perf record` shows readable stacks (dev aid, use with a Release build)" OFF)
# ---- the Rust core (core-rs: matrix-sdk) as a static library ----
find_program(VC_CARGO cargo REQUIRED)
if(CMAKE_BUILD_TYPE STREQUAL "Release")
  set(VC_RUST_PROFILE release)
  set(VC_RUST_TARGET_DIR ${CMAKE_BINARY_DIR}/core-rs)
  set(VC_RUST_OUT ${VC_RUST_TARGET_DIR}/release)
else()
  # development builds share the workspace's target directory (its dependencies are already compiled there)
  set(VC_RUST_PROFILE dev)
  set(VC_RUST_TARGET_DIR ${CMAKE_SOURCE_DIR}/target)
  set(VC_RUST_OUT ${VC_RUST_TARGET_DIR}/debug)
endif()
set(VC_RUST_FEATURES "")
if(VC_DEMO)
  set(VC_RUST_FEATURES --features testkit)
endif()
set(VC_RUST_ENV CARGO_TARGET_DIR=${VC_RUST_TARGET_DIR})
if(VC_CONSOLE)
  # tokio-console: its own target directory (tokio_unstable changes every dependency), see README "Diagnosing stalls"
  set(VC_RUST_TARGET_DIR ${CMAKE_BINARY_DIR}/core-rs)
  if(VC_RUST_PROFILE STREQUAL "dev")
    set(VC_RUST_OUT ${VC_RUST_TARGET_DIR}/debug)
  else()
    set(VC_RUST_OUT ${VC_RUST_TARGET_DIR}/release)
  endif()
  set(VC_RUST_ENV CARGO_TARGET_DIR=${VC_RUST_TARGET_DIR} "RUSTFLAGS=--cfg tokio_unstable")
  set(VC_RUST_FEATURES ${VC_RUST_FEATURES} --features console)
endif()
if(VC_PROFILE)
  # its own target directory: the frame-pointer flag changes every dependency
  set(VC_RUST_TARGET_DIR ${CMAKE_BINARY_DIR}/core-rs)
  set(VC_RUST_OUT ${VC_RUST_TARGET_DIR}/release)
  set(VC_RUST_ENV CARGO_TARGET_DIR=${VC_RUST_TARGET_DIR} "RUSTFLAGS=-C force-frame-pointers=yes" CARGO_PROFILE_RELEASE_DEBUG=1 CARGO_PROFILE_RELEASE_STRIP=false)
  add_compile_options(-g -fno-omit-frame-pointer)
endif()
set(VC_RUST_LIB ${VC_RUST_OUT}/libvector_core.a)
file(GLOB_RECURSE VC_RUST_SOURCES CONFIGURE_DEPENDS core-rs/src/*.rs core-rs/Cargo.toml Cargo.toml Cargo.lock)
add_custom_command(OUTPUT ${VC_RUST_LIB}
  # RUST_MIN_STACK: rustc 1.99 crashed (SIGSEGV in LLVM's inliner) on matrix-sdk-sqlite with codegen-units=1 and the default 8 MB stack
  COMMAND ${CMAKE_COMMAND} -E env ${VC_RUST_ENV} RUST_MIN_STACK=134217728 ${VC_CARGO} build --profile ${VC_RUST_PROFILE} --lib --locked -p vector-core ${VC_RUST_FEATURES} --manifest-path ${CMAKE_SOURCE_DIR}/Cargo.toml
  DEPENDS ${VC_RUST_SOURCES} COMMENT "Building the Rust core (cargo)" VERBATIM)
add_custom_target(vc_rust_build DEPENDS ${VC_RUST_LIB})
add_library(vc_rust STATIC IMPORTED GLOBAL)
set_target_properties(vc_rust PROPERTIES IMPORTED_LOCATION ${VC_RUST_LIB})
add_dependencies(vc_rust vc_rust_build)
target_link_libraries(vc_rust INTERFACE pthread dl m)
# voice calls: the Rust core links libopus and, on Linux, ALSA (sound card)
find_package(PkgConfig REQUIRED)
if(UNIX AND NOT APPLE)
  pkg_check_modules(VC_AUDIO REQUIRED IMPORTED_TARGET alsa opus vpx)
else()
  pkg_check_modules(VC_AUDIO REQUIRED IMPORTED_TARGET opus vpx)
endif()
target_link_libraries(vc_rust INTERFACE PkgConfig::VC_AUDIO)

# ---- the Qt6 desktop app ----
set(VC_QT_COMPONENTS Widgets Multimedia MultimediaWidgets)
if(UNIX AND NOT APPLE)
  list(APPEND VC_QT_COMPONENTS DBus) # desktop notifications (freedesktop.org)
endif()
find_package(Qt6 REQUIRED COMPONENTS ${VC_QT_COMPONENTS})

set(VC_QT_SOURCES
  qt/main.cpp qt/core.cpp qt/theme.cpp qt/uiscale.cpp qt/avatar.cpp qt/imageload.cpp qt/videoplayer.cpp qt/audioplayer.cpp qt/loginpage.cpp qt/sidebar.cpp qt/emoji.cpp qt/emojipicker.cpp qt/packpicker.cpp
  qt/timelineview.cpp qt/notifier.cpp qt/memberlist.cpp qt/searchpanel.cpp qt/dialogs.cpp qt/conversations.cpp qt/mainwindow.cpp)
add_executable(qvector ${VC_QT_SOURCES})
target_include_directories(qvector PRIVATE ${CMAKE_SOURCE_DIR})
target_link_libraries(qvector PRIVATE vc_rust Qt6::Widgets Qt6::Multimedia Qt6::MultimediaWidgets)
if(UNIX AND NOT APPLE)
  target_link_libraries(qvector PRIVATE Qt6::DBus)
endif()
target_compile_options(qvector PRIVATE -Wall -Wextra -Wno-unused-parameter)
if(UNIX AND NOT APPLE)
  # The engine is a static library: without this the linker keeps every function and every vtable of every object file it pulls in, though a third of
  # them are never called. The release program is 90 MB instead of 128 MB and has a smaller footprint (relocations, code pages) when it runs.
  target_link_options(qvector PRIVATE -Wl,--gc-sections)
endif()

include(GNUInstallDirs)
install(TARGETS qvector RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
if(UNIX AND NOT APPLE)
  install(FILES packaging/qvector.desktop DESTINATION ${CMAKE_INSTALL_DATADIR}/applications)
  install(FILES packaging/qvector.png DESTINATION ${CMAKE_INSTALL_DATADIR}/icons/hicolor/256x256/apps)
endif()

# Torture test for the widgets (not built by default): cmake --build build --target qvector_stress
add_executable(qvector_stress EXCLUDE_FROM_ALL qt/stress_main.cpp qt/core.cpp qt/theme.cpp qt/uiscale.cpp qt/avatar.cpp qt/imageload.cpp qt/videoplayer.cpp qt/audioplayer.cpp qt/sidebar.cpp qt/emoji.cpp
  qt/emojipicker.cpp qt/packpicker.cpp qt/timelineview.cpp qt/memberlist.cpp qt/notifier.cpp)
target_include_directories(qvector_stress PRIVATE ${CMAKE_SOURCE_DIR})
target_link_libraries(qvector_stress PRIVATE vc_rust Qt6::Widgets Qt6::Multimedia Qt6::MultimediaWidgets)
# the zoom's key presses are sent with QTest (part of Qt's development files; a build without it only skips that step)
find_package(Qt6 QUIET OPTIONAL_COMPONENTS Test)
if(TARGET Qt6::Test)
  target_link_libraries(qvector_stress PRIVATE Qt6::Test)
  target_compile_definitions(qvector_stress PRIVATE VC_HAVE_QTEST)
endif()
if(UNIX AND NOT APPLE)
  target_link_libraries(qvector_stress PRIVATE Qt6::DBus)
  target_link_options(qvector_stress PRIVATE -Wl,--gc-sections)
endif()
