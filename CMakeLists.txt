cmake_minimum_required(VERSION 3.20)
project(qvector LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)

option(VC_DEMO "Include the fake homeserver so `qvector --demo` runs without an account (dev aid)" ON)

# ---- the Rust core (core-rs: matrix-sdk) as a static library ----
find_program(VC_CARGO cargo REQUIRED)
if(CMAKE_BUILD_TYPE STREQUAL "Release")
  set(VC_RUST_PROFILE release)
  set(VC_RUST_TARGET_DIR ${CMAKE_BINARY_DIR}/core-rs)
  set(VC_RUST_OUT ${VC_RUST_TARGET_DIR}/release)
else()
  # development builds share the workspace's target directory (its dependencies are already compiled there)
  set(VC_RUST_PROFILE dev)
  set(VC_RUST_TARGET_DIR ${CMAKE_SOURCE_DIR}/target)
  set(VC_RUST_OUT ${VC_RUST_TARGET_DIR}/debug)
endif()
set(VC_RUST_FEATURES "")
if(VC_DEMO)
  set(VC_RUST_FEATURES --features testkit)
endif()
set(VC_RUST_LIB ${VC_RUST_OUT}/libvector_core.a)
file(GLOB_RECURSE VC_RUST_SOURCES CONFIGURE_DEPENDS core-rs/src/*.rs core-rs/Cargo.toml Cargo.toml Cargo.lock)
add_custom_command(OUTPUT ${VC_RUST_LIB}
  COMMAND ${CMAKE_COMMAND} -E env CARGO_TARGET_DIR=${VC_RUST_TARGET_DIR} ${VC_CARGO} build --profile ${VC_RUST_PROFILE} --lib --locked -p vector-core ${VC_RUST_FEATURES} --manifest-path ${CMAKE_SOURCE_DIR}/Cargo.toml
  DEPENDS ${VC_RUST_SOURCES} COMMENT "Building the Rust core (cargo)" VERBATIM)
add_custom_target(vc_rust_build DEPENDS ${VC_RUST_LIB})
add_library(vc_rust STATIC IMPORTED GLOBAL)
set_target_properties(vc_rust PROPERTIES IMPORTED_LOCATION ${VC_RUST_LIB})
add_dependencies(vc_rust vc_rust_build)
target_link_libraries(vc_rust INTERFACE pthread dl m)
# voice calls: the Rust core links libopus and, on Linux, ALSA (sound card)
find_package(PkgConfig REQUIRED)
if(UNIX AND NOT APPLE)
  pkg_check_modules(VC_AUDIO REQUIRED IMPORTED_TARGET alsa opus vpx)
else()
  pkg_check_modules(VC_AUDIO REQUIRED IMPORTED_TARGET opus vpx)
endif()
target_link_libraries(vc_rust INTERFACE PkgConfig::VC_AUDIO)

# ---- the Qt6 desktop app ----
set(VC_QT_COMPONENTS Widgets Multimedia MultimediaWidgets)
if(UNIX AND NOT APPLE)
  list(APPEND VC_QT_COMPONENTS DBus) # desktop notifications (freedesktop.org)
endif()
find_package(Qt6 REQUIRED COMPONENTS ${VC_QT_COMPONENTS})

set(VC_QT_SOURCES
  qt/main.cpp qt/core.cpp qt/theme.cpp qt/avatar.cpp qt/videoplayer.cpp qt/audioplayer.cpp qt/loginpage.cpp qt/sidebar.cpp qt/emoji.cpp qt/emojipicker.cpp qt/packpicker.cpp
  qt/timelineview.cpp qt/notifier.cpp qt/memberlist.cpp qt/searchpanel.cpp qt/dialogs.cpp qt/conversations.cpp qt/mainwindow.cpp)
add_executable(qvector ${VC_QT_SOURCES})
target_include_directories(qvector PRIVATE ${CMAKE_SOURCE_DIR})
target_link_libraries(qvector PRIVATE vc_rust Qt6::Widgets Qt6::Multimedia Qt6::MultimediaWidgets)
if(UNIX AND NOT APPLE)
  target_link_libraries(qvector PRIVATE Qt6::DBus)
endif()
target_compile_options(qvector PRIVATE -Wall -Wextra -Wno-unused-parameter)

include(GNUInstallDirs)
install(TARGETS qvector RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
if(UNIX AND NOT APPLE)
  install(FILES packaging/qvector.desktop DESTINATION ${CMAKE_INSTALL_DATADIR}/applications)
  install(FILES packaging/qvector.png DESTINATION ${CMAKE_INSTALL_DATADIR}/icons/hicolor/256x256/apps)
endif()
